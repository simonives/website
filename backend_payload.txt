/**
 * Simon Ives | Strategic Briefing Backend V2 (Hardened)
 *
 * This version uses standard parameter extraction for maximum compatibility
 * with static HTML form submissions.
 */

function doPost(e) {
  // 1. Locate the target spreadsheet
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // 2. Implement a lock to prevent concurrent write failures
  const lock = LockService.getScriptLock();
  
  try {
    lock.waitLock(10000); // 10 second timeout
    
    // 3. Extract parameters from the form-encoded payload
    const params = e.parameter;
    
    // 4. Append row: Timestamp, Name, Email, Scale, Tech, Challenge
    sheet.appendRow([
      new Date(),
      params.name || "Missing Name",
      params.email || "Missing Email",
      params.headcount || "Missing Scale",
      params.technology || "Missing Tech",
      params.request || "Missing Parameters"
    ]);
    
    // 5. Return success signal
    return ContentService
      .createTextOutput("SUCCESS")
      .setMimeType(ContentService.MimeType.TEXT);
      
  } catch (err) {
    return ContentService
      .createTextOutput("ERROR: " + err.toString())
      .setMimeType(ContentService.MimeType.TEXT);
      
  } finally {
    lock.releaseLock();
  }
}

/**
 * Health check endpoint
 */
function doGet(e) {
  return ContentService
    .createTextOutput("System Active.")
    .setMimeType(ContentService.MimeType.TEXT);
}
